  wallet
}

enum RidePaymentStatus {
  pending
  held
  paid
  refunded
}

model DriverProfile {
  id            String       @id @default(cuid())
  userId        String       @unique
  licenseNumber String
  status        DriverStatus @default(pending)
  vehicleClass  String       @default("economy")
  vehicleMake   String
  vehicleModel  String
  vehiclePlate  String
  active        Boolean      @default(true)
  createdAt     DateTime     @default(now())
  updatedAt     DateTime     @updatedAt
  user          User         @relation(fields: [userId], references: [id])
  rides         Ride[]

  @@index([status, active])
}

model Ride {
  id                 String            @id @default(cuid())
  rideNumber         String            @unique
  riderId            String
  driverId           String?
  status             RideStatus        @default(requested)
  pickupLabel        String
  pickupAddress      String
  pickupLat          Float
  pickupLng          Float
  dropoffLabel       String
  dropoffAddress     String
  dropoffLat         Float
  dropoffLng         Float
  vehicleClass       String
  distanceMeters     Int
  durationMinutes    Int
  fareKobo           BigInt
  currency           String            @default("NGN")
  paymentMethod      RidePaymentMethod
  paymentStatus      RidePaymentStatus @default(pending)
  paymentReference   String?
  requestedAt        DateTime          @default(now())
  acceptedAt         DateTime?
  arrivedAt          DateTime?
  startedAt          DateTime?
  completedAt        DateTime?
  cancelledAt        DateTime?
  cancellationReason String?
  createdAt          DateTime          @default(now())
  updatedAt          DateTime          @updatedAt
  rider              User              @relation("RiderRides", fields: [riderId], references: [id])
  driver             DriverProfile?    @relation(fields: [driverId], references: [id])
  events             RideEvent[]

  @@index([riderId, status])
  @@index([driverId, status])
  @@index([status, createdAt])
}

model RideEvent {
  id          String     @id @default(cuid())
  rideId      String
  status      RideStatus
  label       String
  description String
  actorId     String?
  createdAt   DateTime   @default(now())
  ride        Ride       @relation(fields: [rideId], references: [id], onDelete: Cascade)

  @@index([rideId, createdAt])
}
